static void ssa(sqlite3_context *context, int argc, sqlite3_value **argv)
{
    if (argc == 1) {
        double angle = sqlite3_value_double(argv[0]);
        //angle = std::fmod( angle + M_PI, 2 * M_PI ) - M_PI; 
        angle = angle + M_PI;
        const double tau = 2*M_PI;
        angle = (angle - std::floor(angle/tau)*tau) - M_PI;
        sqlite3_result_double(context, angle);
        return;
    }
    sqlite3_result_null(context);
}
